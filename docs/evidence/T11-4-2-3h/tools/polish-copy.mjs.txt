import assert from 'node:assert/strict';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {captureFingerprint,changedInputs} from '../../scripts/release-fingerprint.mjs';
const d='docs/evidence/T11-4-2-3h/',out=d+'polish',sha=b=>createHash('sha256').update(b).digest('hex');
const coverage=JSON.parse(await readFile(d+'release/summary.json','utf8'));assert(coverage.coverageAllPassed);assert.equal(coverage.cases,126);assert.equal(coverage.currentReplacementCases,17);const terminal=JSON.parse(await readFile(d+'release/terminal.json','utf8'));assert.equal(terminal.exitCode,1);assert.equal(terminal.stats.unexpected,5);
const baseline=JSON.parse(await readFile(d+'release/geometry-retry/baseline.json','utf8'));assert.deepEqual(changedInputs(baseline,await captureFingerprint(process.cwd(),baseline.extraInputs)),[]);
const edits=[
 {file:'src/presentation-v3/forge-notices.ts',old:'当前身价 ${fmt(b.value)}，仍可交差。继续锤的风险不会因换剑或刷新而消失。',replacement:'那一记后身价 ${fmt(b.value)}，仍可交差。换剑或刷新不会重置风险；再裂会断。'},
 {file:'src/presentation-v3/forge-notices.ts',old:'交出 ${fmt(b.value)}，就能补上今天还差的 ${fmt(missing)}。身价尚在剑上，交差才计入目标。',replacement:'那一记后身价 ${fmt(b.value)}，已够补上当时缺口 ${fmt(missing)}。身价还在剑上，最新报价见交差按钮。'},
 {file:'src/presentation-v3/app.ts',old:'将开启${contractName(modeDraft)}并替换当前v0.4进度；可以先导出。旧v0.2/v0.3档保留。',replacement:'将开启${contractName(modeDraft)}并替换当前炉前进度；可先导出。旧D细锻与B版进度保留。'},
];
const files=new Map();for(const edit of edits){if(!files.has(edit.file)){const before=await readFile(edit.file,'utf8');assert.equal(sha(before),baseline.sourceHashes[edit.file].sha256);files.set(edit.file,{before,after:before});}const data=files.get(edit.file);assert.equal(data.after.split(edit.old).length-1,1);data.after=data.after.replace(edit.old,edit.replacement);}
await mkdir(out,{recursive:false});await mkdir(out+'/before',{recursive:false});const changes=[];
for(const [file,data]of files){await writeFile(out+'/before/'+file.split('/').at(-1)+'.txt',data.before,{flag:'wx'});await writeFile(file,data.after);changes.push({file,beforeSha256:sha(data.before),afterSha256:sha(data.after),literalReplacements:edits.filter(e=>e.file===file)});}
await writeFile(out+'/copy-changes.json',JSON.stringify({at:new Date().toISOString(),scope:'Three presentation strings only; no command, price, input, layout or state logic change. Full126 retains its original source identity.',changes},null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({changedFiles:changes.map(c=>c.file),literalChanges:edits.length}));
